# Dipendenze del worker di post-produzione AI.
# NON installare pacchetti che parlano con API esterne: il worker parla solo
# con l'app e con il servizio vLLM nel cluster.
#
# CUDA: i nodi GPU hanno il driver 550.x (A100), che esegue le wheel CUDA 12
# (12.4 in modo nativo, dalla 12.5 per compatibilita' di versione minore,
# driver >= 525) ma NON CUDA 13. Le wheel di torch e torchcodec pubblicate su
# PyPI sono CUDA 13 (`libcudart.so.13`): tutto il blocco CUDA viene
# dall'indice cu126, con l'etichetta `+cu126` scritta per intero, perche' una
# versione senza etichetta farebbe scegliere a pip la wheel CUDA 13. cu126 e'
# l'unico indice CUDA 12 che ha torch 2.14. Il build si ferma se entra una
# wheel CUDA 13 (vedi Dockerfile.worker).
--extra-index-url https://download.pytorch.org/whl/cu126
torch==2.14.1+cu126
torchvision==0.29.1+cu126
# torchaudio 2.11 e' l'ultima release (il progetto e' in manutenzione) e non
# dichiara una versione di torch. Il worker non usa API di torchaudio;
# WhisperX usa solo `torchaudio.pipelines` (modelli di allineamento), che la
# 2.11 mantiene.
torchaudio==2.11.0+cu126
# Lettura dell'audio per pyannote.audio 4 (ha sostituito l'I/O di torchaudio).
torchcodec==0.17.0+cu126
numpy==2.5.3

# Client HTTP verso /api/internal/* dell'app e verso vLLM nel cluster. Basta
# l'API sincrona: un worker esegue un job alla volta (un job per pod).
httpx==0.28.1

# Pydantic valida la forma della risposta del claim, cosi' un payload
# malformato non porta a dereferenziare `None`. Corrisponde a quello che
# impone `lib/ai/schemas.ts` lato server.
pydantic==2.9.2

# ASR e diarization. WhisperX (requirements-whisperx.txt) si installa SENZA
# le sue dipendenze: ogni release 3.8.x fissa torch alla 2.8, che non ha le
# correzioni di sicurezza. Le sue dipendenze stanno quindi qui, a versioni
# che funzionano con il torch fissato sopra.
pyannote.audio==4.0.7
faster-whisper==1.2.1
# CTranslate2 dalla 4.6.3 non chiede piu' cuDNN (le wheel sono costruite
# senza): il backend di faster-whisper usa solo cuBLAS 12.
ctranslate2==4.8.3
nltk==3.10.3
omegaconf==2.4.0
pandas==2.3.3

# Sottotitoli VTT, scritti dagli helper in worker/vtt.py.
webvtt-py==0.5.1

# Il caricamento degli artefatti usa URL PUT prefirmati: nessun SDK.

# Modelli Hugging Face per l'allineamento di WhisperX (wav2vec2) e pipeline.
# transformers >= 5.10.1 (la 5.10.0 e' stata ritirata da PyPI) chiude gli
# advisory sul codice di generazione personalizzato e su `save_pretrained`;
# la linea 5.x chiede huggingface-hub >= 1.5, che accetta solo `token=` (mai
# `use_auth_token=`).
transformers==5.19.0
huggingface-hub==2.2.0

# TTS — Piper (rhasspy/piper-tts). MIT license, voci pre-trained
# scaricate da huggingface.co/rhasspy/piper-voices. Gira su CPU
# (ONNX runtime), niente GPU richiesta. Per DUB job.
# Dalla 1.3 il pacchetto e' GPL-3.0-or-later (e la 1.8 cambia l'API di sintesi):
# salire e' una decisione di licenza, oltre che un cambio di codice.
piper-tts==1.2.0

# Sentry / structured logging optional — we use stdlib logging
# configured for JSON. Cluster Loki picks them up.
gender-guesser>=0.4,<1

# AudioSeal (Meta, MIT) — watermark inudibile sull'audio doppiato per
# conformità AI Act Art. 50. Gira sul torch fissato sopra (pin esatti
# vincono sulle dipendenze >= di audioseal). Modello scaricato da HF a
# runtime (audioseal_wm_16bits, ~pochi MB) se non e' nella cartella di
# AUDIOSEAL_CACHE_DIR. Best-effort: il DUB non fallisce se il watermark non
# riesce (vedi worker/tts.py).
audioseal==0.1.4
